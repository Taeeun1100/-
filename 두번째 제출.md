```c
#include "raylib.h"

#define MAX_BULLETS 50
#define MAX_ENEMIES 20

typedef struct {
    Vector2 pos;
    bool active;
} Bullet;

typedef struct {
    Vector2 pos;
    bool active;
} Enemy;

int main(void)
{
    const int screenWidth = 800;
    const int screenHeight = 600;

    InitWindow(screenWidth, screenHeight, "Mini Shooter");
    SetTargetFPS(60);

    Rectangle player = {370, 540, 60, 30};

    Bullet bullets[MAX_BULLETS] = {0};
    Enemy enemies[MAX_ENEMIES] = {0};

    int score = 0;
    bool gameOver = false;
    int spawnTimer = 0;

    while (!WindowShouldClose())
    {
        if (!gameOver)
        {
            // 플레이어 이동
            if (IsKeyDown(KEY_LEFT))
                player.x -= 5;

            if (IsKeyDown(KEY_RIGHT))
                player.x += 5;

            if (player.x < 0)
                player.x = 0;

            if (player.x > screenWidth - player.width)
                player.x = screenWidth - player.width;

            // 총알 발사
            if (IsKeyPressed(KEY_SPACE))
            {
                for (int i = 0; i < MAX_BULLETS; i++)
                {
                    if (!bullets[i].active)
                    {
                        bullets[i].active = true;
                        bullets[i].pos.x = player.x + player.width / 2;
                        bullets[i].pos.y = player.y;
                        break;
                    }
                }
            }

            // 총알 이동
            for (int i = 0; i < MAX_BULLETS; i++)
            {
                if (bullets[i].active)
                {
                    bullets[i].pos.y -= 8;

                    if (bullets[i].pos.y < 0)
                        bullets[i].active = false;
                }
            }

            // 적 생성
            spawnTimer++;

            if (spawnTimer >= 40)
            {
                spawnTimer = 0;

                for (int i = 0; i < MAX_ENEMIES; i++)
                {
                    if (!enemies[i].active)
                    {
                        enemies[i].active = true;
                        enemies[i].pos.x = GetRandomValue(20, screenWidth - 20);
                        enemies[i].pos.y = -20;
                        break;
                    }
                }
            }

            // 적 이동
            for (int i = 0; i < MAX_ENEMIES; i++)
            {
                if (enemies[i].active)
                {
                    enemies[i].pos.y += 3;

                    if (enemies[i].pos.y > screenHeight)
                    {
                        gameOver = true;
                    }
                }
            }

            // 충돌 판정
            for (int i = 0; i < MAX_BULLETS; i++)
            {
                if (!bullets[i].active)
                    continue;

                for (int j = 0; j < MAX_ENEMIES; j++)
                {
                    if (!enemies[j].active)
                        continue;

                    if (CheckCollisionCircles(
                            bullets[i].pos, 5,
                            enemies[j].pos, 15))
                    {
                        bullets[i].active = false;
                        enemies[j].active = false;
                        score++;
                    }
                }
            }
        }

        BeginDrawing();

        ClearBackground(BLACK);

        DrawRectangleRec(player, BLUE);

        for (int i = 0; i < MAX_BULLETS; i++)
        {
            if (bullets[i].active)
                DrawCircleV(bullets[i].pos, 5, YELLOW);
        }

        for (int i = 0; i < MAX_ENEMIES; i++)
        {
            if (enemies[i].active)
                DrawCircleV(enemies[i].pos, 15, RED);
        }

        DrawText(TextFormat("Score : %d", score),
                 10, 10, 30, WHITE);

        if (gameOver)
        {
            DrawText("GAME OVER",
                     250, 250,
                     50, RED);
        }

        EndDrawing();
    }

    CloseWindow();
    return 0;
}
```
